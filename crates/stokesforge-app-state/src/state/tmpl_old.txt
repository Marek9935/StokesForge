// -- Templates: rules in plain sentences --------------------------------------

pub fn template_rules(t: &InstrumentTemplate) -> Vec<String> {
    let r = &t.recipe;
    let mut rules = Vec::new();
    let marker_text = r.main_until_marker.clone();
    if r.subheader_roles.x_prime.is_some() {
        match &marker_text {
            Some(m) => rules.push(tr!("import-template-rule-xprime", marker = m.clone())),
            None => rules.push(tr!("import-template-rule-xprime-plain")),
        }
    } else if let Some(m) = &marker_text {
        rules.push(tr!("import-template-rule-data-after", marker = m.clone()));
    }
    let unit_key = |target: PropTarget| {
        r.header_props
            .iter()
            .find(|p| {
                p.field == PropField::Units && (p.target == target || p.target == PropTarget::All)
            })
            .map(|p| p.key.clone())
    };
    if let (Some(x), Some(y)) = (unit_key(PropTarget::X), unit_key(PropTarget::Y)) {
        rules.push(tr!("import-template-rule-units", x = x, y = y));
    }
    let long_y = r
        .header_props
        .iter()
        .find(|p| p.field == PropField::LongName && p.target == PropTarget::Y);
    match (&r.name_source, long_y) {
        (NameSource::HeaderKey(k), None) => {
            rules.push(tr!("import-template-rule-name-sheet", key = k.clone()))
        }
        (_, Some(p)) => rules.push(tr!("import-template-rule-name-all", key = p.key.clone())),
        _ => {}
    }
    let decimal = if r.decimal == DecimalStyle::Comma {
        tr!("import-check-decimal-comma")
    } else {
        tr!("import-check-decimal-point")
    };
    let sep = match &r.separator {
        stokesforge_import::Separator::Semicolon => sep_name(SepKind::Semicolon),
        stokesforge_import::Separator::Tab => sep_name(SepKind::Tab),
        stokesforge_import::Separator::Comma => sep_name(SepKind::Comma),
        _ => sep_name(SepKind::Whitespace),
    };
    rules.push(tr!("import-template-rule-numbers", decimal = decimal, sep = sep));
    if let Some(m) = &marker_text {
        rules.push(tr!("import-template-rule-marker", marker = m.clone()));
    }
    rules
}

fn today() -> String {
    let secs = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .map_or(0, |d| d.as_secs());
    let days = (secs / 86_400) as i64;
    // Civil date from days since 1970-01-01 (Howard Hinnant's algorithm).
    let z = days + 719_468;
    let era = z.div_euclid(146_097);
    let doe = z.rem_euclid(146_097);
    let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365;
    let y = yoe + era * 400;
    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
    let mp = (5 * doy + 2) / 153;
    let d = doy - (153 * mp + 2) / 5 + 1;
    let m = if mp < 10 { mp + 3 } else { mp - 9 };
    let y = if m <= 2 { y + 1 } else { y };
    format!("{y:04}-{m:02}-{d:02}")
}


#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum TriggerKind {
    Origin,
    DataType,
    Marker,
    Numbers,
    Kinetic,
}

impl TriggerKind {
    pub const ALL: [TriggerKind; 5] = [
        TriggerKind::Origin,
        TriggerKind::DataType,
        TriggerKind::Marker,
        TriggerKind::Numbers,
        TriggerKind::Kinetic,
    ];
}

/// A condition the user added to a template (mockup J): what the automatic
/// ones cannot tell, such as 2 columns against 241 for a JASCO 2D spectrum
/// against a kinetic.
#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub enum Condition {
    /// The number of cells per data line.
    Columns { at_least: bool, n: usize },
    /// A `KEY;value` line of the header.
    HeaderKey {
        key: String,
        value: String,
        contains: bool,
    },
    /// The name of the file contains a text (case ignored).
    FileName { text: String },
    /// The extension of the file, without the dot (case ignored).
    Extension { ext: String },
}

impl Condition {
    /// A plain sentence continuing « Used for files that ».
    pub fn sentence(&self) -> String {
        match self {
            Condition::Columns { at_least: false, n } => {
                tr!("import-condition-columns-eq", n = n.to_string())
            }
            Condition::Columns { at_least: true, n } => {
                tr!("import-condition-columns-min", n = n.to_string())
            }
            Condition::HeaderKey {
                key,
                value,
                contains: false,
            } => tr!(
                "import-condition-key-eq",
                key = key.clone(),
                value = value.clone()
            ),
            Condition::HeaderKey {
                key,
                value,
                contains: true,
            } => tr!(
                "import-condition-key-has",
                key = key.clone(),
                value = value.clone()
            ),
            Condition::FileName { text } => tr!("import-condition-name", text = text.clone()),
            Condition::Extension { ext } => tr!("import-condition-ext", ext = ext.clone()),
        }
    }

    /// Does the file meet it?
    pub fn holds(&self, fp: &Fingerprint) -> bool {
        match self {
            Condition::Columns { at_least, n } => {
                if *at_least {
                    fp.n_cols >= *n
                } else {
                    fp.n_cols == *n
                }
            }
            Condition::HeaderKey {
                key,
                value,
                contains,
            } => fp.keys.iter().any(|(k, v)| {
                k.trim().eq_ignore_ascii_case(key.trim())
                    && if *contains {
                        v.to_lowercase().contains(&value.trim().to_lowercase())
                    } else {
                        v.trim().eq_ignore_ascii_case(value.trim())
                    }
            }),
            Condition::FileName { text } => {
                !text.is_empty() && fp.file_name.to_lowercase().contains(&text.to_lowercase())
            }
            Condition::Extension { ext } => fp
                .file_name
                .rsplit_once('.')
                .is_some_and(|(_, e)| e.eq_ignore_ascii_case(ext.trim().trim_start_matches('.'))),
        }
    }
}

/// The conditions a file must meet to be read with a template. The values are
/// those of the file the template was saved from; `on` says which count.
/// **Never a line count**: the header length varies between files of one
/// instrument.
#[derive(Debug, Clone, PartialEq, Default, Serialize, Deserialize)]
pub struct Triggers {
    pub origin: String,
    pub data_type: String,
    pub marker: String,
    pub sep: String,
    pub comma: bool,
    pub kinetic: bool,
    pub on: [bool; 5],
    /// The conditions the user added.
    #[serde(default)]
    pub custom: Vec<Condition>,
}

fn trigger_index(kind: TriggerKind) -> usize {
    match kind {
        TriggerKind::Origin => 0,
        TriggerKind::DataType => 1,
        TriggerKind::Marker => 2,
        TriggerKind::Numbers => 3,
        TriggerKind::Kinetic => 4,
    }
}

impl Triggers {
    /// Every trigger the file offers, all switched on.
    pub fn from_fingerprint(fp: &Fingerprint) -> Self {
        let mut triggers = Self {
            origin: fp.origin.clone(),
            data_type: fp.dt_key.clone(),
            marker: fp.marker.clone().unwrap_or_default().to_uppercase(),
            sep: fp.sep_symbol.clone(),
            comma: fp.comma,
            kinetic: fp.kinetic,
            on: [false; 5],
            custom: Vec::new(),
        };
        for kind in TriggerKind::ALL {
            let offered = triggers.offered(kind);
            triggers.set(kind, offered);
        }
        triggers
    }

    /// Does the file give this trigger at all?
    pub fn offered(&self, kind: TriggerKind) -> bool {
        match kind {
            TriggerKind::Origin => !self.origin.is_empty(),
            TriggerKind::DataType => !self.data_type.is_empty(),
            TriggerKind::Marker => !self.marker.is_empty(),
            TriggerKind::Numbers => true,
            TriggerKind::Kinetic => self.kinetic,
        }
    }

    pub fn is_on(&self, kind: TriggerKind) -> bool {
        self.on[trigger_index(kind)]
    }

    pub fn set(&mut self, kind: TriggerKind, on: bool) {
        self.on[trigger_index(kind)] = on;
    }

    pub fn any_on(&self) -> bool {
        !self.custom.is_empty()
            || TriggerKind::ALL
                .into_iter()
                .any(|kind| self.offered(kind) && self.is_on(kind))
    }

    /// The conditions of a file, plain sentences (« come from JASCO V-770 »).
    pub fn sentence(&self, kind: TriggerKind) -> String {
        match kind {
            TriggerKind::Origin => tr!("import-condition-auto-origin", name = self.origin.clone()),
            TriggerKind::DataType => tr!(
                "import-condition-auto-type",
                name = short_data_type(&self.data_type)
            ),
            TriggerKind::Marker => tr!("import-condition-auto-marker", marker = self.marker.clone()),
            TriggerKind::Numbers => {
                if self.comma {
                    tr!("import-condition-auto-numbers-comma", sep = self.sep.clone())
                } else {
                    tr!("import-condition-auto-numbers-point", sep = self.sep.clone())
                }
            }
            TriggerKind::Kinetic => {
                if self.marker.is_empty() {
                    tr!("import-condition-auto-kinetic-plain")
                } else {
                    tr!("import-condition-auto-kinetic", marker = self.marker.clone())
                }
            }
        }
    }

    /// Does the file meet every condition that is switched on?
    pub fn matches(&self, fp: &Fingerprint) -> bool {
        if !self.any_on() || (!fp.recognised && self.custom.is_empty()) {
            return false;
        }
        if !self.custom.iter().all(|c| c.holds(fp)) {
            return false;
        }
        TriggerKind::ALL.into_iter().all(|kind| {
            if !(self.offered(kind) && self.is_on(kind)) {
                return true;
            }
            match kind {
                TriggerKind::Origin => self.origin.eq_ignore_ascii_case(&fp.origin),
                TriggerKind::DataType => self.data_type == fp.dt_key,
                TriggerKind::Marker => self
                    .marker
                    .eq_ignore_ascii_case(&fp.marker.clone().unwrap_or_default()),
                TriggerKind::Numbers => self.sep == fp.sep_symbol && self.comma == fp.comma,
                TriggerKind::Kinetic => fp.kinetic,
            }
        })
    }
}

/// A recipe the user asked to remember for an instrument.
#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub struct InstrumentTemplate {
    pub name: String,
    pub id: String,
    pub recipe: ImportRecipe,
    pub decided: Vec<CheckKind>,
    pub created: String,
    pub files: usize,
    #[serde(default)]
    pub triggers: Triggers,
}

/// A template offered for the current file.
#[derive(Debug, Clone)]
pub struct Offer {
    pub name: String,
    pub display: String,
    pub rules: Vec<String>,
}

/// A template as listed in the template menu and the manager.
#[derive(Debug, Clone)]
pub struct TemplateRow {
    pub name: String,
    pub files: usize,
    pub rules: Vec<String>,
    pub in_use: bool,
    pub triggers: Triggers,
}

/// The « Save as Template » window being filled in (mockup J).
#[derive(Debug, Clone, PartialEq)]
pub struct SaveDraft {
    pub name: String,
    pub triggers: Triggers,
    pub rules: Vec<String>,
}

/// Why « Save » is greyed out.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SaveIssue {
    NoName,
    NoTrigger,
}

/// The « Import Templates » window (mockup J).
#[derive(Debug, Clone, Default, PartialEq)]
pub struct ManageState {
    pub selected: Option<String>,
    /// The name being typed; applied on Enter.
    pub rename: String,
}

/// The review in progress: the doubts, in order.
#[derive(Debug, Clone, Default)]

    // -- Instruments -----------------------------------------------------------

    /// The offer to save a template, when the file has a recognised instrument,
    /// no template matches it, nothing is left to check and the offer was not
    /// declined (mockup J).
    pub fn check_offer(&self) -> Option<Offer> {
        let state = self.import_state.as_ref()?;
        if state.checks.review.is_some() {
            return None;
        }
        let path = self.import_preview_path()?;
        let view = self.import_check()?;
        let fp = &view.fingerprint;
        if !fp.recognised
            || view.n_check > 0
            || state.checks.offer_dismissed.contains(&fp.id)
            || state.checks.template_of_file.contains_key(&path)
            || state.checks.template_matching(fp).is_some()
        {
            return None;
        }
        let preview = InstrumentTemplate {
            name: fp.template_name.clone(),
            id: fp.id.clone(),
            recipe: state.recipe.clone(),
            decided: Vec::new(),
            created: today(),
            files: 1,
            triggers: Triggers::from_fingerprint(fp),
        };
        Some(Offer {
            name: fp.template_name.clone(),
            display: fp.origin.clone(),
            rules: template_rules(&preview),
        })
    }

    pub fn check_offer_dismiss(&mut self) {
        let Some(fp) = self.import_fingerprint() else {
            return;
        };
        if let Some(state) = self.import_state.as_mut() {
            state.checks.offer_dismissed.insert(fp.id.clone());
            state.checks.review_done = None;
        }
    }

    // -- Save as Template ------------------------------------------------------

    /// Opens « Save as Template » for the file shown: the proposed name and
    /// every trigger of the file, all checked.
    pub fn check_save_open(&mut self) {
        let Some(view) = self.import_check() else {
            return;
        };
        let Some(state) = self.import_state.as_mut() else {
            return;
        };
        let fp = &view.fingerprint;
        let preview = InstrumentTemplate {
            name: fp.template_name.clone(),
            id: fp.id.clone(),
            recipe: state.recipe.clone(),
            decided: Vec::new(),
            created: today(),
            files: 1,
            triggers: Triggers::default(),
        };
        state.checks.save_draft = Some(SaveDraft {
            name: fp.template_name.clone(),
            triggers: Triggers::from_fingerprint(fp),
            rules: template_rules(&preview),
        });
    }

    pub fn check_save_cancel(&mut self) {
        if let Some(state) = self.import_state.as_mut() {
            state.checks.save_draft = None;
        }
    }

    pub fn check_save_draft(&self) -> Option<&SaveDraft> {
        self.import_state.as_ref()?.checks.save_draft.as_ref()
    }

    pub fn check_save_draft_mut(&mut self) -> Option<&mut SaveDraft> {
        self.import_state.as_mut()?.checks.save_draft.as_mut()
    }

    /// Why « Save » is greyed out, if it is.
    pub fn check_save_issue(&self) -> Option<SaveIssue> {
        let draft = self.check_save_draft()?;
        if draft.name.trim().is_empty() {
            Some(SaveIssue::NoName)
        } else if !draft.triggers.any_on() {
            Some(SaveIssue::NoTrigger)
        } else {
            None
        }
    }

    /// Would « Save » replace a template of the same name?
    pub fn check_save_replaces(&self) -> bool {
        let Some(draft) = self.check_save_draft() else {
            return false;
        };
        self.import_state
            .as_ref()
            .is_some_and(|s| s.checks.template_named(draft.name.trim()).is_some())
    }

    /// « Save »: keeps the recipe, what was settled and the triggers.
    pub fn check_save_commit(&mut self) {
        if self.check_save_issue().is_some() {
            return;
        }
        let Some(view) = self.import_check() else {
            return;
        };
        let Some((path, key)) = self.check_kind_key() else {
            return;
        };
        let fp = view.fingerprint.clone();
        let Some(state) = self.import_state.as_mut() else {
            return;
        };
        let Some(draft) = state.checks.save_draft.take() else {
            return;
        };
        let name = draft.name.trim().to_string();
        let recipe = state.recipe.clone();
        let mut decided: Vec<CheckKind> = state
            .checks
            .decided
            .get(&key)
            .map(|d| d.keys().copied().collect())
            .unwrap_or_default();
        for f in &view.findings {
            if f.confidence == Confidence::Check && !decided.contains(&f.kind) {
                decided.push(f.kind);
            }
        }
        let files = state.checks.template_named(&name).map_or(1, |t| t.files + 1);
        let template = InstrumentTemplate {
            name: name.clone(),
            id: fp.id.clone(),
            recipe,
            decided: decided.clone(),
            created: today(),
            files,
            triggers: draft.triggers,
        };
        state.checks.templates.retain(|t| t.name != name);
        state.checks.templates.insert(0, template);
        state.checks.save_templates();
        state.checks.templates_generation += 1;
        state.checks.template_of_file.insert(path, name.clone());
        let map = state.checks.decided.entry(key).or_default();
        for kind in decided {
            map.insert(kind, Source::Template(name.clone()));
        }
        state.checks.offer_dismissed.insert(fp.id.clone());
        state.checks.review_done = None;
        state.checks.generation += 1;
    }

    // -- Import Templates --------------------------------------------------------

    pub fn check_manage_open(&mut self) {
        let first = self
            .import_state
            .as_ref()
            .and_then(|s| s.checks.templates.first().map(|t| t.name.clone()));
        if let Some(state) = self.import_state.as_mut() {
            state.checks.manage = Some(ManageState {
                rename: first.clone().unwrap_or_default(),
                selected: first,
            });
        }
    }

    pub fn check_manage_close(&mut self) {
        if let Some(state) = self.import_state.as_mut() {
            state.checks.manage = None;
            state.checks.forget_pending = None;
        }
    }

    pub fn check_manage(&self) -> Option<&ManageState> {
        self.import_state.as_ref()?.checks.manage.as_ref()
    }

    pub fn check_manage_mut(&mut self) -> Option<&mut ManageState> {
        self.import_state.as_mut()?.checks.manage.as_mut()
    }

    pub fn check_manage_select(&mut self, name: &str) {
        if let Some(manage) = self.check_manage_mut() {
            manage.selected = Some(name.to_string());
            manage.rename = name.to_string();
        }
    }

    /// Renames a template (applied on Enter); an empty name or one that
    /// another template carries is refused.
    pub fn check_rename_template(&mut self, old: &str, new: &str) {
        let new = new.trim().to_string();
        let Some(state) = self.import_state.as_mut() else {
            return;
        };
        if new.is_empty() || new == old || state.checks.template_named(&new).is_some() {
            return;
        }
        for t in &mut state.checks.templates {
            if t.name == old {
                t.name = new.clone();
            }
        }
        for name in state.checks.template_of_file.values_mut() {
            if name == old {
                *name = new.clone();
            }
        }
        if let Some(manage) = state.checks.manage.as_mut() {
            manage.selected = Some(new);
        }
        state.checks.save_templates();
        state.checks.templates_generation += 1;
        state.checks.generation += 1;
    }

    /// A trigger switched on or off, applied live.
    pub fn check_set_trigger(&mut self, name: &str, kind: TriggerKind, on: bool) {
        let Some(state) = self.import_state.as_mut() else {
            return;
        };
        // Never switch the last one off: a template that triggers on nothing
        // would match no file.
        if let Some(t) = state.checks.templates.iter_mut().find(|t| t.name == name) {
            if !on {
                let mut probe = t.triggers.clone();
                probe.set(kind, false);
                if !probe.any_on() {
                    return;
                }
            }
            t.triggers.set(kind, on);
        }
        state.checks.save_templates();
        state.checks.templates_generation += 1;
        state.checks.generation += 1;
    }

    /// The conditions the user added to a template, replaced live. Refused
    /// when the template would trigger on nothing.
    pub fn check_set_conditions(&mut self, name: &str, custom: Vec<Condition>) {
        let Some(state) = self.import_state.as_mut() else {
            return;
        };
        if let Some(t) = state.checks.templates.iter_mut().find(|t| t.name == name) {
            let mut probe = t.triggers.clone();
            probe.custom = custom.clone();
            if !probe.any_on() {
                return;
            }
            if t.triggers.custom == custom {
                return;
            }
            t.triggers.custom = custom;
        }
        state.checks.save_templates();
        state.checks.templates_generation += 1;
        state.checks.generation += 1;
    }

    pub fn check_forget_ask(&mut self, name: &str) {
        if let Some(state) = self.import_state.as_mut() {
            state.checks.forget_pending = Some(name.to_string());
        }
    }

    pub fn check_forget_pending(&self) -> Option<&str> {
        self.import_state.as_ref()?.checks.forget_pending.as_deref()
    }

    pub fn check_forget_cancel(&mut self) {
        if let Some(state) = self.import_state.as_mut() {
            state.checks.forget_pending = None;
        }
    }

    /// Confirmed: forgets the pending template and selects the next one.
    pub fn check_forget_confirm(&mut self) {
        let Some(name) = self.check_forget_pending().map(str::to_owned) else {
            return;
        };
        self.check_forget_template(&name);
        let next = self
            .import_state
            .as_ref()
            .and_then(|s| s.checks.templates.first().map(|t| t.name.clone()));
        if let Some(state) = self.import_state.as_mut() {
            state.checks.forget_pending = None;
            if let Some(manage) = state.checks.manage.as_mut() {
                manage.rename = next.clone().unwrap_or_default();
                manage.selected = next;
            }
        }
    }

    /// Reads the file shown with a remembered template.
    pub fn check_use_template(&mut self, name: &str) {
        let Some((path, key)) = self.check_kind_key() else {
            return;
        };
        let bytes = self.import_bytes(&path).ok();
        let Some(state) = self.import_state.as_mut() else {
            return;
        };
        let Some(template) = state.checks.template_named(name).cloned() else {
            return;
        };
        let mut recipe = template.recipe.clone();
        if let Some(bytes) = &bytes {
            if let Some(main) = stokesforge_import::sniff::marker_main_lines(bytes, &recipe) {
                recipe.header_layout.main_header_lines = main;
            }
        }
        state.recipe = recipe;
        let map = state.checks.decided.entry(key).or_default();
        for kind in &template.decided {
            map.insert(*kind, Source::Template(template.name.clone()));
        }
        state.checks.auto_paths.remove(&path);
        state.checks.template_of_file.insert(path, template.name);
        state.checks.generation += 1;
    }

    /// « Auto »: forget the template of the file shown, detect again.
    pub fn check_use_auto(&mut self) {
        let Some((path, key)) = self.check_kind_key() else {
            return;
        };
        if let Some(state) = self.import_state.as_mut() {
            state.checks.template_of_file.remove(&path);
            state.checks.auto_paths.insert(path.clone());
            state.checks.decided.remove(&key);
            state.checks.generation += 1;
            state.checks.review = None;
            state.checks.review_done = None;
        }
        self.import_auto_redetect();
    }

    /// « Forget » a remembered instrument.
    pub fn check_forget_template(&mut self, name: &str) {
        if let Some(state) = self.import_state.as_mut() {
            state.checks.templates.retain(|t| t.name != name);
            state.checks.template_of_file.retain(|_, n| n != name);
            state.checks.save_templates();
            state.checks.templates_generation += 1;
            state.checks.generation += 1;
        }
    }

    pub fn check_template_rows(&self) -> Vec<TemplateRow> {
        let Some(state) = self.import_state.as_ref() else {
            return Vec::new();
        };
        let path = self.import_preview_path();
        let in_use = path
            .as_ref()
            .and_then(|p| state.checks.template_of_file.get(p));
        state
            .checks
            .templates
            .iter()
            .map(|t| TemplateRow {
                name: t.name.clone(),
                files: t.files,
                rules: template_rules(t),
                in_use: in_use == Some(&t.name),
                triggers: t.triggers.clone(),
            })
            .collect()
    }

    /// Name of the template the file shown is read with.
    pub fn check_template_in_use(&self) -> Option<String> {
        let state = self.import_state.as_ref()?;
        let path = self.import_preview_path()?;
        state.checks.template_of_file.get(&path).cloned()
    }

    /// Picks the template of an instrument that is already remembered, when
    /// a file arrives (called by the detector glue). Returns the recipe to
    /// read the file with.
    pub(super) fn check_template_recipe(
        checks: &mut ImportChecks,
        path: &Path,
        bytes: &[u8],
        found: &Sniff,
    ) -> Option<ImportRecipe> {
        let fp = checks.fingerprint(path, bytes, found)?;
        if checks.auto_paths.contains(path) {
            return None;
        }
        let template = checks.template_matching(&fp)?.clone();
        let mut recipe = template.recipe.clone();
        if let Some(main) = stokesforge_import::sniff::marker_main_lines(bytes, &recipe) {
            recipe.header_layout.main_header_lines = main;
        }
        let map = checks.decided.entry(found.kind.clone()).or_default();
        for kind in &template.decided {
            map.insert(*kind, Source::Template(template.name.clone()));
        }
        trust
            .template_of_file
            .insert(path.to_path_buf(), template.name.clone());
        checks.generation += 1;
        Some(recipe)
    }
