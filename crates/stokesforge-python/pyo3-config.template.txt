# PyO3 configuration template for the embedded interpreter (pitfall no. 1).
#
# python-build-standalone exposes a sysconfig whose LIBDIR is /install/lib
# (a path on Astral's build machine, nonexistent here): PyO3's automatic
# detection fails or points to nothing. So we bypass the interpreter
# query entirely by providing this file through the
# PYO3_CONFIG_FILE environment variable.
#
# The substitution token below is replaced by scripts/python-config.sh,
# which writes the
# real file to target/pyo3-config.txt (not versioned: the absolute path
# depends on the machine). This script also removes the comment lines:
# the pyo3-build-config parser expects only key=value pairs.
implementation=CPython
version=3.12
shared=true
abi3=false
lib_name=python3.12
lib_dir=@PYTHON_PREFIX@/lib
executable=@PYTHON_PREFIX@/bin/python3
pointer_width=64
build_flags=
suppress_build_script_link_lines=false
