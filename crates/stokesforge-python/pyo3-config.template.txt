# Gabarit de configuration PyO3 pour l'interpréteur embarqué (piège n°1).
#
# python-build-standalone expose un sysconfig dont LIBDIR vaut /install/lib
# (chemin de la machine de build d'Astral, inexistant ici) : la détection
# automatique de PyO3 échoue ou pointe dans le vide. On court-circuite donc
# entièrement l'interrogation de l'interpréteur en fournissant ce fichier via
# la variable d'environnement PYO3_CONFIG_FILE.
#
# Le jeton de substitution ci-dessous est remplacé par scripts/python-config.sh,
# qui écrit le
# fichier réel dans target/pyo3-config.txt (non versionné : le chemin absolu
# dépend de la machine). Ce script retire aussi les lignes de commentaire :
# le parseur de pyo3-build-config n'attend que des paires clé=valeur.
implementation=CPython
version=3.12
shared=true
abi3=false
lib_name=python3.12
lib_dir=@PYTHON_PREFIX@/lib
executable=@PYTHON_PREFIX@/bin/python3
pointer_width=64
build_flags=
suppress_build_script_link_lines=false
